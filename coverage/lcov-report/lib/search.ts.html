// ── Turkish Normalization ──────────────────────────────
 
export function normalizeTurkish(text: string): string {
    return text
        .replace(/İ/g, 'I')
        .toLowerCase()
        .replace(/ı/g, 'i')
        .replace(/ö/g, 'o')
        .replace(/Ö/g, 'o')
        .replace(/ü/g, 'u')
        .replace(/Ü/g, 'u')
        .replace(/ç/g, 'c')
        .replace(/Ç/g, 'c')
        .replace(/ş/g, 's')
        .replace(/Ş/g, 's')
        .replace(/ğ/g, 'g')
        .replace(/Ğ/g, 'g');
}
 
function matches(haystack: string, needle: string): boolean {
    return normalizeTurkish(haystack).includes(normalizeTurkish(needle));
}
